def segment(source, name, next_marker, private=True):
    marker = '    private fun ' if private else '    fun '
    start = source.index(marker + name + '(')
    return source[start:source.index(next_marker, start)]


def kotlin_delimiters_balanced(source):
    """Conservative lexical delimiter check only, not Kotlin compilation or type checking."""
    stack = []
    index = 0
    pairs = {')': '(', ']': '[', '}': '{'}
    while index < len(source):
        if source.startswith('//', index):
            end = source.find('\n', index + 2)
            index = len(source) if end < 0 else end + 1
        elif source.startswith('/*', index):
            depth = 1
            index += 2
            while depth and index < len(source):
                if source.startswith('/*', index):
                    depth += 1
                    index += 2
                elif source.startswith('*/', index):
                    depth -= 1
                    index += 2
                else:
                    index += 1
            if depth:
                return False
        elif source.startswith('"""', index):
            end = source.find('"""', index + 3)
            if end < 0:
                return False
            index = end + 3
        elif source[index] in ('"', "'"):
            quote = source[index]
            index += 1
            while index < len(source):
                if source[index] == '\\':
                    index += 2
                elif source[index] == quote:
                    index += 1
                    break
                else:
                    index += 1
            else:
                return False
        else:
            char = source[index]
            if char in '([{':
                stack.append(char)
            elif char in ')]}':
                if not stack or stack.pop() != pairs[char]:
                    return False
            index += 1
    return not stack
