def digest(data):
    return hashlib.sha256(data).hexdigest()


def source_segment(source, name, next_marker, private=True):
    marker = '    private fun ' if private else '    fun '
    start = source.index(marker + name + '(')
    return source[start:source.index(next_marker, start)]


def literal_constant(path, name):
    """Read one literal source constant via AST; never import/exec a candidate test module."""
    tree = ast.parse(path.read_text(encoding='utf-8'))
    matches = [node.value for node in tree.body if isinstance(node, ast.Assign)
               and any(isinstance(target, ast.Name) and target.id == name for target in node.targets)]
    if len(matches) != 1:
        raise ValueError('Expected one literal source constant: ' + name)
    return ast.literal_eval(matches[0])


def reconstruct_exact_successor(data, successor_digest, predecessor_digest, exact_inserted_bytes, exact_old_bytes):
    """A named one-occurrence correction only; unrelated mutations may not disappear."""
    if digest(data) != successor_digest or data.count(exact_inserted_bytes) != 1:
        raise ValueError('Not the exact reviewed successor and insertion')
    predecessor = data.replace(exact_inserted_bytes, exact_old_bytes, 1)
    if digest(predecessor) != predecessor_digest:
        raise ValueError('Unrelated bytes changed outside the exact compatibility correction')
    return predecessor

    def test_thirty_five_kotlin_and_source_data_controls_do_not_create_processes_authority_or_execution_receipts(self):
        kotlin_names = re.findall(r'@Test fun (\w+)\(', self.kotlin_tests)
        self.assertEqual(35, len(kotlin_names))
        self.assertEqual(35, len(set(kotlin_names)))
        for name in ('mandatoryUnmappedOperatorCompanionCannotDisappearChangeSeverityMoveOrAcquireInventedMapping',
                     'sixHardErrorSuppressionsAlwaysStayUnknownWithEveryActualFailureAndWarningRetained',
                     'locationsRequireActualAuthoredDigestPathPointAndNarrowSourceRegionNotAnyValidFilePosition',
                     'parserValueControlsKeepGlobalWarningsMalformedStatusAndQuotedNamesSeparateFromEmissions',
                     'privateCompletionOverlayAndLatestSidecarNeverGrantOrdinaryMatrixOrQualificationCredit'):
            self.assertIn(name, kotlin_names)
        # Match actual complete authority type tokens, not the suffix of the unchanged @Test method name.
        # This is a source-control lexical check, not a general Kotlin parser or security boundary.
        authority_types = ('ExecutionAuthority', 'ExactExecutionAuthority',
                           'ExactDetektModelExecutionAuthority', 'ExactDetektFullModelExecutionAuthority')
        authority_constructor = r'(?<![\w$])(?:' + '|'.join(authority_types) + r')\s*(?:\(|\{)'
        other_forbidden = ('ProcessOutcome(', 'ProcessBuilder(', 'KotlinJvmCompilerProvider(', '.analyze(', '.analyzeObserved(',
                           'Runtime.getRuntime(', 'Files.write', 'Files.create', 'Files.delete', 'ExecutionPermit(')

        def assert_no_execution_surface(source):
            for forbidden in other_forbidden:
                self.assertNotIn(forbidden, source)
            self.assertNotRegex(source, authority_constructor)

        assert_no_execution_surface(self.plan + self.kotlin_tests)
        assert_no_execution_surface('@Test fun optionOffSourceIdentityAndExactFlagChangeAreNotArbitraryManifestExecutionAuthority() {')
        for authority in authority_types:
            for prefix in ('', 'dev.ktchecker.example.'):
                for opening in ('(', ' (', '\t(', '\n(', '{', ' {', '\n{'):
                    with self.assertRaises(AssertionError):
                        assert_no_execution_surface(prefix + authority + opening)
        for forbidden in other_forbidden:
            with self.assertRaises(AssertionError):
                assert_no_execution_surface(forbidden)
        self.assertIn('No actual compiler/provider/native process or receipt is created or simulated', self.kotlin_tests)
        self.assertIn('unit_value_only_not_run', self.kotlin_tests)
        for filename, expected_count in (('test_diagnostic_breadth.py', 16), ('test_diagnostic_breadth_hookup.py', 18)):
            source = (FIXTURES / filename).read_text(encoding='utf-8')
            tree = ast.parse(source)
            names = [node.name for node in ast.walk(tree) if isinstance(node, ast.FunctionDef) and node.name.startswith('test_')]
            self.assertEqual(expected_count, len(names))
            imports = {alias.name.split('.')[0] for node in ast.walk(tree) if isinstance(node, ast.Import) for alias in node.names}
            imports |= {node.module.split('.')[0] for node in ast.walk(tree) if isinstance(node, ast.ImportFrom) and node.module}
            self.assertFalse(imports & {'subprocess', 'socket', 'urllib', 'requests', 'importlib', 'nullability_protocols'})
            for call in [node for node in ast.walk(tree) if isinstance(node, ast.Call) and isinstance(node.func, ast.Name)]:
                self.assertNotIn(call.func.id, ('eval', 'exec', 'compile', '__import__'))
